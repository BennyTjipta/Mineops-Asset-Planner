cmake_minimum_required(VERSION 3.16)
project(MineOpsAssetPlanner VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(assetops src/main.cpp)
if(MSVC)
    target_compile_options(assetops PRIVATE /W4)
else()
    target_compile_options(assetops PRIVATE -Wall -Wextra -Wpedantic)
endif()
